// ------------------------------------------ // Salvar e Carregar Projetos Cosplay (localStorage) // ------------------------------------------ // Chave do banco local const STORAGE_KEY = "nebulab_cosplay_projects"; // Carregar projetos salvos ao abrir a página document.addEventListener("DOMContentLoaded", () => { loadSavedProjects(); }); // Função para salvar um projeto function saveProject(projectName, character, budget, checklistItems) { const projects = JSON.parse(localStorage.getItem(STORAGE_KEY)) || []; const newProject = { id: Date.now(), name: projectName || "Projeto Sem Nome", character: character || "Não especificado", budget: budget || 0, checklist: checklistItems || [], createdAt: new Date().toLocaleDateString('pt-BR') }; projects.push(newProject); localStorage.setItem(STORAGE_KEY, JSON.stringify(projects)); notify("✦ Projeto salvo com sucesso no NEBULAB!"); loadSavedProjects(); } // Função para listar os projetos salvos no HTML function loadSavedProjects() { const container = document.getElementById("savedProjectsList"); if (!container) return; const projects = JSON.parse(localStorage.getItem(STORAGE_KEY)) || []; if (projects.length === 0) { container.innerHTML = "

Nenhum projeto salvo no seu terminal local.

"; return; } container.innerHTML = projects.map(proj => `
${proj.name} (${proj.character})
Orçamento: R$ ${proj.budget} | Criado em: ${proj.createdAt}
`).join(""); } // Função para remover um projeto salvo function deleteProject(id) { let projects = JSON.parse(localStorage.getItem(STORAGE_KEY)) || []; projects = projects.filter(p => p.id !== id); localStorage.setItem(STORAGE_KEY, JSON.stringify(projects)); loadSavedProjects(); notify("✦ Projeto removido."); } // ------------------------------------------ // Busca de Materiais e Comparador de Preços // ------------------------------------------ async function searchProducts() { const productInput = document.getElementById("productSearchInput"); const query = productInput ? productInput.value.trim() : ""; const resultsContainer = document.getElementById("productResults"); if (!query) { alert("Digite o nome de um produto ou material (ex: EVA 5mm, Peruca loira, Tinta acrílica)."); return; } if (resultsContainer) { resultsContainer.innerHTML = "

✦ Mapeando lojas e estimando preços...

"; } // Links diretos configurados para pesquisar o produto em grandes e-commerces const searchLinks = { shopee: `https://shopee.com.br/search?keyword=${encodeURIComponent(query)}`, mercadolivre: `https://lista.mercadolivre.com.br/${encodeURIComponent(query)}`, aliexpress: `https://pt.aliexpress.com/w/wholesale-${encodeURIComponent(query)}.html` }; // Exemplo de estimativa de faixas de preço por categoria comum de cosplay let priceEstimate = "R$ 25,00 - R$ 120,00"; if (query.toLowerCase().includes("eva")) priceEstimate = "R$ 15,00 - R$ 45,00 (placa)"; else if (query.toLowerCase().includes("wig") || query.toLowerCase().includes("peruca")) priceEstimate = "R$ 60,00 - R$ 180,00"; else if (query.toLowerCase().includes("lente")) priceEstimate = "R$ 40,00 - R$ 90,00"; setTimeout(() => { if (resultsContainer) { resultsContainer.innerHTML = `

✦ Estimativa Médiana de Mercado: ${priceEstimate}

🛒 Shopee 📦 Mercado Livre 🌐 AliExpress
`; } }, 500); }

/* ========================================== NEBULAB - LÓGICA DE PESQUISA & EXPORTAÇÃO ========================================== */ // ------------------------------------------ // 1. Pesquisa Real de Personagens (API Jikan) // ------------------------------------------ async function searchCharacter() { const searchInput = document.getElementById("searchInput"); const query = searchInput ? searchInput.value.trim() : ""; const grid = document.querySelector(".character-grid"); if (!query) { alert("Por favor, digite o nome de um personagem."); return; } if (grid) { grid.innerHTML = "

✦ Buscando na constelação de dados...

"; } try { const response = await fetch(`https://api.jikan.moe/v4/characters?q=${encodeURIComponent(query)}&limit=6`); const data = await response.json(); if (!data.data || data.data.length === 0) { grid.innerHTML = "

Nenhum personagem encontrado na galáxia.

"; return; } grid.innerHTML = ""; // Limpa os resultados anteriores data.data.forEach(char => { const charCard = document.createElement("div"); charCard.className = "character"; charCard.id = `char-card-${char.mal_id}`; // Pega uma descrição resumida ou texto padrão const aboutText = char.about ? char.about.substring(0, 120).replace(/\n/g, ' ') + "..." : "Sem descrição disponível."; charCard.innerHTML = `
O3 // PERSONAGEM

${char.name}

${aboutText}

`; grid.appendChild(charCard); }); } catch (error) { console.error("Erro na busca:", error); if (grid) { grid.innerHTML = "

Ocorreu um erro ao conectar-se à rede.

"; } } } // Evento para acionar a busca ao apertar "Enter" const searchInput = document.getElementById("searchInput"); if (searchInput) { searchInput.addEventListener("keydown", (e) => { if (e.key === "Enter") searchCharacter(); }); } // ------------------------------------------ // 2. Exportar Elementos como Imagem (PNG) // ------------------------------------------ function downloadCard(elementId, fileName = "nebulab-card") { const element = document.getElementById(elementId); if (!element) return; html2canvas(element, { backgroundColor: "#050713", // Mantém o fundo escuro do tema useCORS: true // Permite carregar imagens de domínios externos }).then(canvas => { const link = document.createElement("a"); link.download = `${fileName.toLowerCase().replace(/\s+/g, '-')}-nebulab.png`; link.href = canvas.toDataURL("image/png"); link.click(); }).catch(err => { console.error("Erro ao gerar imagem:", err); alert("Erro ao exportar a imagem. Tente novamente."); }); } // ------------------------------------------ // 3. Calculadora de Planos & Planilha de Custos Cosplay // ------------------------------------------ let currentBudgetDetails = null; function calculateCosplayBudget() { const budgetInput = document.getElementById("budgetInput"); const periodicitySelect = document.getElementById("periodicitySelect"); // Semanal, Mensal ou Anual const resultDiv = document.getElementById("budgetResult"); const totalAmount = parseFloat(budgetInput ? budgetInput.value : 0); const periodicity = periodicitySelect ? periodicitySelect.value : "Mensal"; if (isNaN(totalAmount) || totalAmount <= 0) { alert("Por favor, insira um valor válido de orçamento."); return; } // Proporções estimadas para montagem do cosplay const clothes = totalAmount * 0.30; const materials = totalAmount * 0.25; const wig = totalAmount * 0.15; const accessories = totalAmount * 0.15; const makeup = totalAmount * 0.10; const reserve = totalAmount * 0.05; // Guarda os detalhes para exportar para CSV depois currentBudgetDetails = { totalAmount, periodicity, clothes, materials, wig, accessories, makeup, reserve }; const money = (val) => val.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' }); if (resultDiv) { resultDiv.innerHTML = ` ✦ Planejamento Financeiro (${periodicity})

👕 Roupa/Base (30%): ${money(clothes)}
🧰 Materiais/Props (25%): ${money(materials)}
💇 Wig/Peruca (15%): ${money(wig)}
🎭 Acessórios (15%): ${money(accessories)}
💄 Caracterização (10%): ${money(makeup)}
✦ Reserva de Emergência (5%): ${money(reserve)}

`; } } // Exporta o orçamento como arquivo CSV (compatível com Excel) function exportBudgetToCSV() { if (!currentBudgetDetails) return; const b = currentBudgetDetails; let csvContent = "data:text/csv;charset=utf-8,"; csvContent += "Categoria,Porcentagem,Valor Planejado (BRL),Plano\n"; csvContent += `Roupa e Traje,30%,${b.clothes.toFixed(2)},${b.periodicity}\n`; csvContent += `Materiais e Prop,25%,${b.materials.toFixed(2)},${b.periodicity}\n`; csvContent += `Wig e Cabelo,15%,${b.wig.toFixed(2)},${b.periodicity}\n`; csvContent += `Acessorios,15%,${b.accessories.toFixed(2)},${b.periodicity}\n`; csvContent += `Maquiagem e Lentes,10%,${b.makeup.toFixed(2)},${b.periodicity}\n`; csvContent += `Reserva de Emergencia,5%,${b.reserve.toFixed(2)},${b.periodicity}\n`; csvContent += `TOTAL,100%,${b.totalAmount.toFixed(2)},${b.periodicity}\n`; const encodedUri = encodeURI(csvContent); const link = document.createElement("a"); link.setAttribute("href", encodedUri); link.setAttribute("download", `planilha-cosplay-${b.periodicity.toLowerCase()}.csv`); document.body.appendChild(link); link.click(); document.body.removeChild(link); }
